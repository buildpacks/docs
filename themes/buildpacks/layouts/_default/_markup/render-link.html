{{- /* Renders markdown links. Links that point to a different host than the site's own (i.e. external links) open in a new tab, get safe rel attributes, and are followed by a small arrow icon so readers know they will leave the site. The arrow is kept together with the last word of the link text inside a nowrap span, so it can never wrap onto a line by itself. */ -}} {{- $isExternal := false -}} {{- with urls.Parse .Destination -}} {{- if and .Host (ne .Host (urls.Parse site.BaseURL).Host) -}} {{- $isExternal = true -}} {{- end -}} {{- end -}} {{- if not $isExternal -}} {{ .Text | safeHTML }} {{- else -}} {{- $icon := partial "external-link-icon.html" . | chomp -}} {{- $text := .Text -}} {{- $inner := "" -}} {{- if and (not (strings.Contains $text "<")) (strings.Contains $text " ") -}} {{- $words := split $text " " -}} {{- $last := index $words (sub (len $words) 1) -}} {{- $head := delimit (first (sub (len $words) 1) $words) " " -}} {{- $inner = printf "%s %s%s" $head $last $icon -}} {{- else -}} {{- $inner = printf "%s%s" $text $icon -}} {{- end -}} {{ $inner | safeHTML }} {{- end -}}
